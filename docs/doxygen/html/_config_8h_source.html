13/// Narrow outcomes permitted from transport callbacks.
14///
15/// Driver lifecycle or scheduling states such as NOT_INITIALIZED, BUSY, and
16/// IN_PROGRESS cannot cross the transport boundary.
17enum class TransportErr : uint8_t {
18 OK = 0, ///< Complete transaction, including STOP, succeeded
19 NACK_ADDR, ///< Address phase was not acknowledged
20 NACK_DATA, ///< Data phase was not acknowledged
21 TIMEOUT, ///< Physical transfer exceeded its deadline
22 BUS, ///< Arbitration, stuck-line, or controller/bus fault
23 OTHER ///< Truthful failure with no narrower exposed cause
24};
25
26/// Result returned by an injected I2C transport.
27struct TransportStatus {
28 TransportErr code = TransportErr::OK; ///< Narrow transport outcome
29 int32_t detail = 0; ///< Optional backend-specific diagnostic value
30
31 constexpr TransportStatus() = default;
32
33 /// Construct an explicit transport result.
34 /// @param error Narrow transport outcome.
35 /// @param detailCode Optional backend-specific diagnostic value.
36 constexpr TransportStatus(TransportErr error, int32_t detailCode)
37 : code(error), detail(detailCode) {}
38
39 /// Test whether the complete physical transaction succeeded.
40 /// @return true only when code is TransportErr::OK.
41 constexpr bool ok() const { return code == TransportErr::OK; }
42
43 /// Create a successful transport result.
44 /// @return Transport result with code OK and detail zero.
45 static constexpr TransportStatus Ok() { return {}; }
46
47 /// Create a failed transport result.
48 /// @param error Narrow truthful failure code to store.
49 /// @param detailCode Optional backend-specific diagnostic value.
50 /// @return Transport result containing the supplied failure information.
51 static constexpr TransportStatus Error(TransportErr error,
52 int32_t detailCode = 0) {
53 return TransportStatus{error, detailCode};
54 }
55};

57/// I2C write callback signature.
58///
59/// This driver calls the callback with exactly one control byte. A successful
60/// return MUST mean that the complete I2C transaction, including its
61/// terminating STOP condition, has completed. The callback must return within
62/// `timeoutMs` and must preserve the narrow outcome in TransportStatus (for
63/// example address NACK, data NACK, timeout, or bus error). The driver never
64/// retries a transfer.
65/// @param addr I2C device address (7-bit)
66/// @param data Pointer to data to write
67/// @param len Number of bytes to write
68/// @param timeoutMs Maximum time to wait for completion
69/// @param user User context pointer passed through from Config
70/// @return Narrow transport result
71using I2cWriteFn = TransportStatus (*)(uint8_t addr, const uint8_t* data,
72 size_t len, uint32_t timeoutMs,
73 void* user);
74
75/// I2C write-read callback signature.
76///
77/// The TCA9548A control-byte read is requested with `txData == nullptr`,
78/// `txLen == 0`, and `rxLen == 1`; it is one read-only I2C transaction. A
79/// successful return MUST mean that the complete transaction and terminating
80/// STOP condition have completed. The callback must return within `timeoutMs`
81/// and preserve the narrow transport outcome in TransportStatus. The driver
82/// never retries a transfer.
83/// @param addr I2C device address (7-bit)
84/// @param txData Pointer to data to write (nullptr if txLen == 0)
85/// @param txLen Number of bytes to write (0 for read-only)
86/// @param rxData Pointer to buffer for read data
87/// @param rxLen Number of bytes to read
88/// @param timeoutMs Maximum time to wait for completion
89/// @param user User context pointer passed through from Config
90/// @return Narrow transport result
91using I2cWriteReadFn = TransportStatus (*)(uint8_t addr,
92 const uint8_t* txData, size_t txLen,
93 uint8_t* rxData, size_t rxLen,
94 uint32_t timeoutMs, void* user);
95
96/// Optional hard-reset callback for the active-low RESET pin.
97/// Implementations must assert RESET low for at least cmd::RESET_MIN_LOW_NS,
98/// release it, and return only after RESET is complete. The callback is invoked
99/// at most once per hardReset() call and must itself have a finite, documented
100/// execution bound; the driver performs no delay or retry.
101/// @param timeoutMs Maximum time allowed for the complete RESET operation
102/// @param user User context pointer (Config::resetUser)
103/// @return Status indicating success or failure
104using HardResetFn = Status (*)(uint32_t timeoutMs, void* user);
105
106/// Optional millisecond timestamp callback used only for passive diagnostics.
107/// It must be monotonic, nonblocking, bounded, and perform no bus I/O.
108/// @param user User context pointer passed through from Config
109/// @return Current monotonic milliseconds
110using NowMsFn = uint32_t (*)(void* user);

112/// Configuration for TCA9548A driver.
113///
114/// begin() copies this value. The callback targets and context pointers named
115/// by that copy are non-owning and must remain valid until end(). The
116/// application owns bus configuration, serialization, locking, timeout
117/// enforcement, retry, and recovery policy.
118struct Config {
119 // === I2C Transport (required) ===
120 I2cWriteFn i2cWrite = nullptr; ///< I2C write function pointer
121 I2cWriteReadFn i2cWriteRead = nullptr; ///< I2C write-read function pointer
122 void* i2cUser = nullptr; ///< User context for transport callbacks
123 HardResetFn hardReset = nullptr; ///< Optional hardware reset callback
124 void* resetUser = nullptr; ///< User context for hardReset callback
125
126 // === Timing Hooks (optional) ===
127 NowMsFn nowMs = nullptr; ///< Monotonic millisecond source
128 void* timeUser = nullptr; ///< User context for timing hooks
129
130 // === Device Settings ===
131 uint8_t i2cAddress = cmd::DEFAULT_ADDRESS; ///< I2C address: 0x70-0x77
132 uint32_t i2cTimeoutMs = 50; ///< I2C timeout in ms (1..60000)
133 uint32_t resetTimeoutMs = 10; ///< RESET callback timeout in ms (1..60000)
134
135 // === Passive Health Classification ===
136 /// Consecutive failures at which state() reports OFFLINE (1..255). This
137 /// diagnostic classification never prevents a transport operation.
138 uint8_t offlineThreshold = 5;
139};
