57/// Return true when `address` is in the TCA9548A 7-bit address range.
58/// @param address Candidate 7-bit I2C address.
59/// @return true for addresses 0x70 through 0x77; otherwise false.
60constexpr bool isValidAddress(uint8_t address) {
61 return address >= I2C_ADDR_MIN && address <= I2C_ADDR_MAX;
62}

64/// Encode the datasheet A2/A1/A0 pin levels as a 7-bit I2C address.
65/// @param a2 true when address pin A2 is tied high.
66/// @param a1 true when address pin A1 is tied high.
67/// @param a0 true when address pin A0 is tied high.
68/// @return Encoded 7-bit address in the range 0x70 through 0x77.
69constexpr uint8_t addressFromPins(bool a2, bool a1, bool a0) {
70 return static_cast<uint8_t>(
71 I2C_ADDR_BASE | (a2 ? 0x04U : 0x00U) | (a1 ? 0x02U : 0x00U) |
72 (a0 ? 0x01U : 0x00U));
73}
