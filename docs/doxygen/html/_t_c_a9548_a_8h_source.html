27/// Fixed-size value type for the TCA9548A control byte.
28class ChannelMask {
29public:
30 /// Construct the safe all-disabled mask.
31 constexpr ChannelMask() = default;
32
33 /// Construct the safe all-disabled mask.
34 /// @return Mask value 0x00.
35 static constexpr ChannelMask none() { return ChannelMask{cmd::NO_CHANNELS}; }
36
37 /// Construct the all-enabled mask.
38 /// @return Mask value 0xFF.
39 static constexpr ChannelMask all() { return ChannelMask{cmd::ALL_CHANNELS}; }
40
41 /// Return a one-hot mask. An out-of-range cast of Channel produces none().
42 /// @param channel Typed downstream channel.
43 /// @return One-hot mask, or none() for an invalid cast value.
44 static constexpr ChannelMask one(Channel channel) {
45 return static_cast<uint8_t>(channel) < cmd::NUM_CHANNELS
46 ? ChannelMask{static_cast<uint8_t>(
47 1U << static_cast<uint8_t>(channel))}
48 : none();
49 }
50
51 /// Explicit raw conversion for legitimate multi-channel masks.
52 /// @param rawMask Complete control-byte value.
53 /// @return Mask containing exactly rawMask.
54 static constexpr ChannelMask fromRaw(uint8_t rawMask) {
55 return ChannelMask{rawMask};
56 }
57
58 /// Return the encoded control byte.
59 /// @return Raw 8-bit channel mask.
60 constexpr uint8_t raw() const { return _value; }
61
62 /// Test whether a channel is enabled in this value.
63 /// @param channel Typed downstream channel.
64 /// @return true when the channel is valid and its bit is set.
65 constexpr bool contains(Channel channel) const {
66 return static_cast<uint8_t>(channel) < cmd::NUM_CHANNELS &&
67 (_value & one(channel)._value) != 0;
68 }
69
70 /// Test whether all channels are disabled.
71 /// @return true only for mask value 0x00.
72 constexpr bool isNone() const { return _value == cmd::NO_CHANNELS; }
73
74 /// Test whether exactly one channel is enabled.
75 /// @return true only for nonzero masks containing one set bit.
76 constexpr bool isOneHot() const {
77 return _value != 0 && (_value & static_cast<uint8_t>(_value - 1U)) == 0;
78 }
79
80 /// Return this mask with additional channels enabled.
81 /// @param channels Bits to enable.
82 /// @return Bitwise union of this mask and channels.
83 constexpr ChannelMask withEnabled(ChannelMask channels) const {
84 return fromRaw(static_cast<uint8_t>(_value | channels._value));
85 }
86
87 /// Return this mask with selected channels disabled.
88 /// @param channels Bits to disable.
89 /// @return This mask with every bit in channels cleared.
90 constexpr ChannelMask withDisabled(ChannelMask channels) const {
91 return fromRaw(
92 static_cast<uint8_t>(_value & static_cast<uint8_t>(~channels._value)));
93 }
94
95private:
96 explicit constexpr ChannelMask(uint8_t value) : _value(value) {}
97 uint8_t _value = cmd::NO_CHANNELS;
98};

103/// Provenance of the cached channel-mask observation.
104enum class MaskProvenance : uint8_t {
105 UNKNOWN, ///< Hardware state may differ from the cached byte
106 WRITE_COMPLETED, ///< Successful write returned after terminating STOP
107 READBACK_OBSERVED ///< Successful control-byte read observed this value
108};

110/// Truthful cached channel-mask value and its provenance.
111struct ChannelMaskObservation {
112 ChannelMask mask = ChannelMask::none(); ///< Last retained control-byte value
113 MaskProvenance provenance =
114 MaskProvenance::UNKNOWN; ///< Evidence supporting mask
115
116 /// Test whether the retained byte has successful hardware evidence.
117 /// @return true for a completed write or observed readback.
118 constexpr bool known() const { return provenance != MaskProvenance::UNKNOWN; }
119
120 /// Test whether the retained byte came from an actual readback.
121 /// @return true only for READBACK_OBSERVED provenance.
122 constexpr bool verified() const {
123 return provenance == MaskProvenance::READBACK_OBSERVED;

127/// Passive driver-health classification.
128///
129/// OFFLINE never blocks an operation; the external I2C owner retains admission,
130/// retry, recovery, and bus-reset authority.
131enum class DriverState : uint8_t {
132 UNINIT, ///< No successful device transaction in the current binding
133 READY, ///< Most recent tracked transport operation succeeded
134 DEGRADED, ///< 1 <= consecutiveFailures < offlineThreshold
135 OFFLINE ///< consecutiveFailures >= offlineThreshold

153/// Managed synchronous TCA9548A driver.
154///
155/// Ownership and bounds:
156/// - The driver is non-owning and never configures, locks, retries, or recovers
157/// the I2C bus.
158/// - Except for hardReset(), each fallible hardware API performs at most one
159/// timeout-bounded transport transaction and never waits or retries.
160/// - hardReset() is an explicit rare-operation exception: it invokes exactly
161/// one reset callback and, if that callback succeeds, exactly one verification
162/// read. Both receive finite configured timeouts, so its maximum callback
163/// budget is resetTimeoutMs + i2cTimeoutMs; there are no waits/retries.
164/// - Operations are synchronous and terminal. The owner may cancel between
165/// calls; an in-flight callback must terminate within its supplied timeout.
166/// - Calls are not thread-safe, reentrant, or ISR-safe. One external bus owner
167/// must serialize driver and callback access.
168class TCA9548A {
169public:
170 /// Construct an unbound driver with zeroed binding-local diagnostics.
171 TCA9548A() = default;
172 TCA9548A(const TCA9548A&) = delete;
173 TCA9548A& operator=(const TCA9548A&) = delete;
174 TCA9548A(TCA9548A&&) = delete;
175 TCA9548A& operator=(TCA9548A&&) = delete;
176
177 /// Validate and bind configuration, then perform one control-byte presence
178 /// read as required by the managed lifecycle contract.
179 ///
180 /// A valid configuration remains bound if the presence read fails, and that
181 /// exact transport Status is returned. The owner may retry probe(),
182 /// readChannelMask(), or another primitive without rebinding. Rebinding is
183 /// rejected with BUSY until end() is called.
184 /// @param config Valid callback, timeout, address, and health configuration.
185 /// @return Presence-read result, or a validation/lifecycle error without I2C.
186 Status begin(const Config& config);
187
188 /// No-op; the device has no pending I/O or state machine.
189 /// @param nowMs Ignored compatibility timestamp.
190 void tick(uint32_t nowMs);
191
192 /// Unbind without bus I/O. Explicitly call and check disableAll() first when
193 /// safe-off is required.
194 void end();
195
196 /// Perform one raw diagnostic control-byte read without health accounting.
197 /// A successful read stores READBACK_OBSERVED provenance. Exact transport
198 /// errors are returned unchanged; the part has no identity register.
199 /// @return Read result, or NOT_INITIALIZED when no Config is bound.
200 Status probe();
201
202 /// Make one explicit safe-off recovery attempt by writing 0x00.
203 /// Performs one transfer, never asserts RESET or restores a previous mask.
204 /// @return Write result, or NOT_INITIALIZED when no Config is bound.
205 Status recover();
206
207 /// Invoke RESET once with resetTimeoutMs and, on callback success, perform one
208 /// verification read with i2cTimeoutMs. Success requires exactly 0x00; no
209 /// prior mask is restored. A mismatch returns RESET_STATE_MISMATCH with the
210 /// observed byte in detail and retains it as READBACK_OBSERVED.
211 /// @return Terminal RESET callback or verification-read result.
212 Status hardReset();
