9/// Error codes for all TCA9548A operations.
10///
11/// Numeric values are an append-only compatibility contract. The synchronous
12/// core does not synthesize DEVICE_NOT_FOUND or IN_PROGRESS; it preserves
13/// specific I2C failures and requires terminal callback results.
14enum class Err : uint8_t {
15 OK = 0, ///< Operation successful
16 NOT_INITIALIZED = 1, ///< No valid Config is currently bound
17 INVALID_CONFIG = 2, ///< Invalid configuration parameter
18 I2C_ERROR = 3, ///< I2C communication failure (unspecified)
19 TIMEOUT = 4, ///< Non-I2C callback timeout (for example RESET)
20 INVALID_PARAM = 5, ///< Invalid parameter value
21 DEVICE_NOT_FOUND = 6, ///< Reserved compatibility device-absence code
22 UNSUPPORTED = 7, ///< Operation not supported (missing callback)
23 I2C_NACK_ADDR = 8, ///< I2C NACK on address
24 I2C_NACK_DATA = 9, ///< I2C NACK on data
25 I2C_TIMEOUT = 10, ///< I2C transaction timeout
26 I2C_BUS = 11, ///< I2C bus error (SDA stuck, arbitration, etc.)
27 BUSY = 12, ///< Operation cannot start because state is busy
28 IN_PROGRESS = 13, ///< Reserved for sibling-driver compatibility
29 RESET_STATE_MISMATCH = 14 ///< RESET completed but byte was not 0x00
30};
31
32/// Status structure returned by all fallible operations
33struct Status {
34 Err code = Err::OK; ///< Stable typed result code
35 int32_t detail = 0; ///< Backend or operation-specific detail
36 const char* msg = ""; ///< Static-lifetime diagnostic string
37
38 constexpr Status() = default;
39
40 /// Construct an explicit status value.
41 /// @param c Stable typed result code.
42 /// @param d Optional backend or operation-specific detail.
43 /// @param m Static-lifetime diagnostic message.
44 constexpr Status(Err c, int32_t d, const char* m)
45 : code(c), detail(d), msg(m) {}
46
47 /// @return true if operation succeeded
48 constexpr bool ok() const { return code == Err::OK; }
49
50 /// @param expected Error code to compare with Status::code.
51 /// @return true if status matches the requested error code
52 constexpr bool is(Err expected) const { return code == expected; }
53
54 /// Test the reserved sibling-driver progress code.
55 /// @return true only when code equals Err::IN_PROGRESS.
56 constexpr bool inProgress() const { return code == Err::IN_PROGRESS; }
57
58 /// @return true if operation succeeded
59 explicit constexpr operator bool() const { return ok(); }
60
61 /// Create a success status.
62 /// @return Status with code OK, detail zero, and a static message.
63 static constexpr Status Ok() { return Status{Err::OK, 0, "OK"}; }
64
65 /// Create an error status.
66 /// @param err Stable typed error code.
67 /// @param message Static-lifetime diagnostic message.
68 /// @param detailCode Optional backend or operation-specific detail.
69 /// @return Status containing the supplied error information.
70 static constexpr Status Error(Err err, const char* message,
71 int32_t detailCode = 0) {
72 return Status{err, detailCode, message};
73 }
74};
